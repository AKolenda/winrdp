cmake_minimum_required(VERSION 3.22)
project(WinRdpNativeBridge VERSION 0.6.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
find_package(Threads REQUIRED)
set(ENGINE "${CMAKE_CURRENT_SOURCE_DIR}/../engine/src")
add_library(winrdp_core STATIC "${ENGINE}/core.cpp")
target_include_directories(winrdp_core PUBLIC "${ENGINE}")
target_link_libraries(winrdp_core PUBLIC Threads::Threads)
if(BUILD_TESTING)
  add_executable(core_tests ../tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE winrdp_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(geometry_tests ../tests/geometry_tests.cpp)
  target_include_directories(geometry_tests PRIVATE "${ENGINE}")
  add_test(NAME geometry COMMAND geometry_tests)
endif()
option(WINRDP_CORE_ONLY "Only build dependency-free tests" OFF)
if(WINRDP_CORE_ONLY)
  return()
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(FREERDP REQUIRED IMPORTED_TARGET freerdp3>=3.31 freerdp-client3 winpr3)
set(CMAKE_AUTOMOC ON)
add_library(winrdp_bridge STATIC bridge.cpp bridge.h
  "${ENGINE}/rdp_session.cpp" "${ENGINE}/rdp_session.hpp"
  "${ENGINE}/profile.cpp" "${ENGINE}/profile.hpp")
target_include_directories(winrdp_bridge PRIVATE "${ENGINE}")
target_link_libraries(winrdp_bridge PUBLIC winrdp_core Qt6::Core Qt6::Gui Qt6::Network
  PkgConfig::GTK3 PkgConfig::FREERDP)
target_compile_options(winrdp_bridge PRIVATE -Wall -Wextra -Wpedantic)
